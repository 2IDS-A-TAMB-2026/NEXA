// ======================================================
// PINOS DO RFID
// ======================================================

#define RFID_SDA  5
#define RFID_RST 22

#define RFID_SCK  18
#define RFID_MISO 19
#define RFID_MOSI 23


// ======================================================
// LEDS E BUZZER

// ======================================================

// Endereço = 0x27
//
// SDA = GPIO 21
// SCL = GPIO 15
//

#define LCD_SCL 15

LiquidCrystal_PCF8574 lcd(0x27);


// ======================================================
// RFID
// ======================================================

MFRC522 leitorRFID(
  RFID_SDA,
  RFID_RST
);


// ======================================================
// DADOS DO FUNCIONÁRIO
// ======================================================

String funcionarioCPF = "";

String setorNome = "";
String setorLocal = "";

String cameraId = "";
String cameraIdentificador = "";


// ======================================================
// LCD - APROXIME O CARTÃO
// ======================================================

void lcdAproximeCartao() {

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("APROXIME O");

  lcd.setCursor(0, 1);
  lcd.print("CARTAO!");
}


// ======================================================
// LCD - PROCESSANDO
// ======================================================

void lcdProcessando() {

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("PROCESSANDO...");

  lcd.setCursor(0, 1);
  lcd.print("AGUARDE");
}


// ======================================================
// LCD - ACESSO PERMITIDO
// ======================================================

void lcdAcessoPermitido() {

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("ACESSO PERMITIDO");

  lcd.setCursor(0, 1);

  if (funcionarioNome.length() <= 16) {

    lcd.print(funcionarioNome);

  } else {

    lcd.print(
      funcionarioNome.substring(0, 16)
    );
  }
}


// ======================================================
// LCD - ACESSO NEGADO
// ======================================================

void lcdAcessoNegado() {

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("ACESSO NEGADO");

  lcd.setCursor(0, 1);
  lcd.print("CARTAO INVALIDO");
}


// ======================================================
// CONECTAR AO WI-FI
// ======================================================

void conectarWifi() {

  Serial.println();
  Serial.print("Conectando ao Wi-Fi");

  WiFi.begin(
    ssid,
    password
  );

  int tentativas = 0;

  while (
    WiFi.status() != WL_CONNECTED
  ) {

    delay(500);

    Serial.print(".");

    tentativas++;

    if (tentativas >= 30) {

      Serial.println();
      Serial.println(
        "Falha ao conectar ao Wi-Fi."
      );

      return;
    }
  }

  Serial.println();
  Serial.println(
    "Wi-Fi conectado!"
  );

  Serial.print(
    "IP do ESP32: "
  );

  Serial.println(
    WiFi.localIP()
  );
}


// ======================================================
// LER UID
// ======================================================

String lerUID() {

  String uid = "";

  for (
    byte i = 0;
    i < leitorRFID.uid.size;
    i++
  ) {

    if (
      leitorRFID.uid.uidByte[i] < 0x10
    ) {

      uid += "0";
    }

    uid += String(
      leitorRFID.uid.uidByte[i],
      HEX
    );

    if (
      i < leitorRFID.uid.size - 1
    ) {

      uid += " ";
    }
  }

  uid.toUpperCase();

  return uid;
}

// ======================================================
// INDICAR SUCESSO
// ======================================================

void indicarSucesso() {

  Serial.println();
  Serial.println(
    "========================================"
  );

  Serial.println(
    "          ACESSO LIBERADO"
  );

  Serial.println(
    "========================================"
  );

  // LCD
  lcdAcessoPermitido();

  // LED verde
  digitalWrite(
    LED_VERDE,
    HIGH
  );

  // LED vermelho desligado
  digitalWrite(
    LED_VERMELHO,
    LOW
  );

  // Buzzer
  tone(
    BUZZER,
    1500
  );

  delay(200);

  noTone(
    BUZZER
  );

  delay(1000);

  digitalWrite(
    LED_VERDE,
    LOW
  );
}

  Serial.println();
  Serial.println(
    "========================================"
  );

  Serial.println(
    "           ACESSO NEGADO"
  );

  Serial.println(
    "========================================"
  );

  // LCD
  lcdAcessoNegado();

  // Verde desligado
  digitalWrite(
    LED_VERDE,
    LOW
  );

  // Vermelho ligado
  digitalWrite(
    LED_VERMELHO,
    HIGH
  );

  // Três bips
  for (
    int i = 0;
    i < 3;
    i++
  ) {

    tone(
      BUZZER,
      500
    );

    delay(200);

    noTone(
      BUZZER
    );

    delay(150);
  }

  digitalWrite(
    LED_VERMELHO,
    LOW
  );
}

// ======================================================
// ENVIAR RFID PARA API
// ======================================================

bool enviarRFID(String uid) {

  // ----------------------------------------------------
  // VERIFICAR WI-FI
  // ----------------------------------------------------

  if (
    WiFi.status() != WL_CONNECTED
  ) {

    Serial.println();
    Serial.println(
      "Wi-Fi desconectado."
    );

    conectarWifi();

    if (
      WiFi.status() != WL_CONNECTED
    ) {

      return false;
    }
  }


  // ----------------------------------------------------
  // LIMPAR DADOS
  // ----------------------------------------------------

  limparDadosFuncionario();


  // ----------------------------------------------------
  // LCD PROCESSANDO
  // ----------------------------------------------------

  lcdProcessando();


  // ----------------------------------------------------
  // HTTP
  // ----------------------------------------------------

  HTTPClient http;

  Serial.println();

  Serial.println(
    "========================================"
  );

  Serial.println(
    "       CONSULTANDO API NEXA"
  );

  Serial.println(
    "========================================"
  );

  Serial.print(
    "URL: "
  );

  Serial.println(
    serverUrl
  );


  http.begin(
    serverUrl
  );


  http.addHeader(
    "Content-Type",
    "application/json"
  );


  // ----------------------------------------------------
  // JSON
  // ----------------------------------------------------

  StaticJsonDocument<256> json;

  json["uid"] = uid;

  String corpoJson;

  serializeJson(
    json,
    corpoJson
  );


  Serial.print(
    "JSON enviado: "
  );

  Serial.println(
    corpoJson
  );


  // ----------------------------------------------------
  // POST
  // ----------------------------------------------------

  int codigoResposta =
    http.POST(
      corpoJson
    );


  Serial.print(
    "Codigo HTTP: "
  );

  Serial.println(
    codigoResposta
  );


  // ----------------------------------------------------
  // ERRO HTTP
  // ----------------------------------------------------

  if (
    codigoResposta <= 0
  ) {

    Serial.print(
      "Erro HTTP: "
    );

    Serial.println(
      http.errorToString(
        codigoResposta
      )
    );

    http.end();

    return false;
  }


  // ----------------------------------------------------
  // RESPOSTA
  // ----------------------------------------------------

  String resposta =
    http.getString();


  Serial.println();

  Serial.println(
    "Resposta da API:"
  );

  Serial.println(
    resposta
  );


  // ----------------------------------------------------
  // PARSE JSON
  // ----------------------------------------------------

  StaticJsonDocument<2048> respostaJson;

  DeserializationError erro =
    deserializeJson(
      respostaJson,
      resposta
    );


  if (
    erro
  ) {

    Serial.println();

    Serial.println(
      "Erro ao interpretar JSON da API."
    );

    http.end();

    return false;
  }


  // ----------------------------------------------------
  // VERIFICAR ACESSO
  // ----------------------------------------------------

  bool sucesso =
    respostaJson["sucesso"] | false;

  bool acesso =
    respostaJson["acesso"] | false;


  // ----------------------------------------------------
  // ACESSO NEGADO
  // ----------------------------------------------------

  if (
    codigoResposta != 200 ||
    !sucesso ||
    !acesso
  ) {

    Serial.println();

    Serial.println(
      "ACESSO NEGADO."
    );


    const char* mensagem =
      respostaJson["mensagem"] |
      "Cartao nao autorizado.";


    Serial.print(
      "Motivo: "
    );

    Serial.println(
      mensagem
    );


    http.end();

    return false;
  }


  // ----------------------------------------------------
  // FUNCIONÁRIO
  // ----------------------------------------------------

  JsonObject funcionario =
    respostaJson["funcionario"];


  funcionarioCPF =
    funcionario["cpf"] | "";

  funcionarioNome =
    funcionario["nome"] | "";

  funcionarioEmail =
    funcionario["email"] | "";


  // ----------------------------------------------------
  // EMPRESA
  // ----------------------------------------------------

  JsonObject empresa =
    respostaJson["empresa"];


  empresaNome =
    empresa["nome"] | "";

  empresaCNPJ =
    empresa["cnpj"] | "";


  // ----------------------------------------------------
  // SETOR
  // ----------------------------------------------------

  JsonObject setor =
    respostaJson["setor"];


  setorNome =
    setor["nome"] | "";

  setorLocal =
    setor["local"] | "";


  // ----------------------------------------------------
  // CÂMERA
  // ----------------------------------------------------

  JsonObject camera =
    respostaJson["camera"];


  if (
    !camera.isNull()
  ) {

    cameraId =
      camera["id"] | "";

    cameraIdentificador =
      camera["identificador"] | "";
  }


  // ----------------------------------------------------
  // MOSTRAR FUNCIONÁRIO
  // ----------------------------------------------------

  mostrarFuncionario();


  // ----------------------------------------------------
  // PRÓXIMA ETAPA
  // ----------------------------------------------------

  JsonObject proximaEtapa =
    respostaJson["proximaEtapa"];


  const char* acao =
    proximaEtapa["acao"] | "";


  Serial.println();

  Serial.println(
    "----------------------------------------"
  );

  Serial.println(
    "PROXIMA ETAPA"
  );

  Serial.println(
    "----------------------------------------"
  );

  Serial.print(
    "Acao: "
  );

  Serial.println(
    acao
  );

  Serial.print(
    "Camera ID: "
  );

  Serial.println(
    cameraId
  );

  Serial.print(
    "Camera: "
  );

  Serial.println(
    cameraIdentificador
  );

  Serial.println(
    "----------------------------------------"
  );


  http.end();


  return true;
}


// ======================================================
// SETUP
// ======================================================

void setup() {

  // ----------------------------------------------------
  // SERIAL
  // ----------------------------------------------------

  Serial.begin(
    115200
  );

  delay(1000);


  // ----------------------------------------------------
  // PINOS
  // ----------------------------------------------------

  pinMode(
    LED_VERDE,
    OUTPUT
  );

  pinMode(
    LED_VERMELHO,
    OUTPUT
  );

  pinMode(
    BUZZER,
    OUTPUT
  );


  digitalWrite(
    LED_VERDE,
    LOW
  );

  digitalWrite(
    LED_VERMELHO,
    LOW
  );


  // ----------------------------------------------------
  // I2C
  // ----------------------------------------------------

  Wire.begin(
    LCD_SDA,
    LCD_SCL
  );


  // ----------------------------------------------------
  // LCD
  // ----------------------------------------------------

  lcd.begin(16, 2);
  lcd.setBacklight(255);

  lcd.clear();

  lcd.setCursor(
    0,
    0
  );

  lcd.print(
    "NEXA"
  );

  lcd.setCursor(
    0,
    1
  );

  lcd.print(
    "Iniciando..."
  );


  delay(2000);


  // ----------------------------------------------------
  // SPI
  // ----------------------------------------------------

  SPI.begin(
    RFID_SCK,
    RFID_MISO,
    RFID_MOSI,
    RFID_SDA
  );


  // ----------------------------------------------------
  // RFID
  // ----------------------------------------------------

  leitorRFID.PCD_Init();

  delay(500);


  // ----------------------------------------------------
  // SERIAL
  // ----------------------------------------------------

  Serial.println();

  Serial.println(
    "========================================"
  );

  Serial.println(
    "             NEXA IoT"
  );

  Serial.println(
    "          LEITOR RFID ESP32"
  );

  Serial.println(
    "========================================"
  );

  Serial.println();

  Serial.println(
    "RFID inicializado."
  );


  // ----------------------------------------------------
  // WI-FI
  // ----------------------------------------------------

  conectarWifi();


  // ----------------------------------------------------
  // SISTEMA PRONTO
  // ----------------------------------------------------

  Serial.println();

  Serial.println(
    "========================================"
  );

  Serial.println(
    "       SISTEMA PRONTO"
  );

  Serial.println(
    "========================================"
  );

  Serial.println(
    "Aproxime o cartao ou tag RFID..."
  );

  Serial.println();


  // ----------------------------------------------------
  // LCD
  // ----------------------------------------------------

  lcdAproximeCartao();
}


// ======================================================
// LOOP
// ======================================================

void loop() {

  // ----------------------------------------------------
  // VERIFICAR WI-FI
  // ----------------------------------------------------

  if (
    WiFi.status() != WL_CONNECTED
  ) {

    Serial.println();

    Serial.println(
      "Wi-Fi desconectado."
    );

    conectarWifi();

    return;
  }


  // ----------------------------------------------------
  // VERIFICAR CARTÃO
  // ----------------------------------------------------

  if (
    !leitorRFID.PICC_IsNewCardPresent()
  ) {

    return;
  }


  // ----------------------------------------------------
  // LER CARTÃO
  // ----------------------------------------------------

  if (
    !leitorRFID.PICC_ReadCardSerial()
  ) {

    return;
  }


  // ----------------------------------------------------
  // UID
  // ----------------------------------------------------

  String uidLido =
    lerUID();


  Serial.println();

  Serial.println(
    "========================================"
  );

  Serial.println(
    "          CARTAO DETECTADO"
  );

  Serial.println(
    "========================================"
  );

  Serial.print(
    "UID: "
  );

  Serial.println(
    uidLido
  );


  // ----------------------------------------------------
  // PROCESSANDO
  // ----------------------------------------------------

  lcdProcessando();

  delay(300);


  // ----------------------------------------------------
  // API
  // ----------------------------------------------------

  bool acessoLiberado =
    enviarRFID(
      uidLido
    );


  // ----------------------------------------------------
  // RESULTADO
  // ----------------------------------------------------

  if (
    acessoLiberado
  ) {

    indicarSucesso();

  } else {

    indicarErro();
  }


  // ----------------------------------------------------
  // FINALIZAR RFID
  // ----------------------------------------------------

  leitorRFID.PICC_HaltA();

  leitorRFID.PCD_StopCrypto1();


  // ----------------------------------------------------
  // EVITAR DUPLICAÇÃO
  // ----------------------------------------------------

  delay(1500);


  // ----------------------------------------------------
  // TELA INICIAL
  // ----------------------------------------------------

  lcdAproximeCartao();


  Serial.println();

  Serial.println(
    "Aproxime outro cartao ou tag..."
  );
}
